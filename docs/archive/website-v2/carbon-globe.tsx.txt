"use client";

import { useEffect, useRef } from "react";

type Vector = [number, number, number];
const radians = Math.PI / 180;
const vector = (longitude: number, latitude: number): Vector => [Math.cos(latitude * radians) * Math.sin(longitude * radians), -Math.sin(latitude * radians), Math.cos(latitude * radians) * Math.cos(longitude * radians)];

export function CarbonGlobe({ paused = false }: { paused?: boolean }) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const pauseRef = useRef(paused);
  useEffect(() => { pauseRef.current = paused; }, [paused]);
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const context = canvas.getContext("2d");
    if (!context) return;
    const controller = new AbortController();
    const media = window.matchMedia("(prefers-reduced-motion: reduce)");
    let coastlines: Vector[][] = [];
    let frame = 0, width = 0, height = 0, active = true, elapsed = 0, lastTime = 0, lastDraw = 0;
    const stars = Array.from({length: 90}, (_, i) => ({x: ((i * 137.508) % 997) / 997, y: ((i * 79.41) % 983) / 983, a: .12 + (i % 5) * .06}));
    const graticule: Vector[][] = [];
    for (let lat = -60; lat <= 60; lat += 20) graticule.push(Array.from({length: 121}, (_, i) => vector(i * 3 - 180, lat)));
    for (let lon = -180; lon < 180; lon += 20) graticule.push(Array.from({length: 91}, (_, i) => vector(lon, i * 2 - 90)));
    const origins = [[-8.5, 33.25], [2.35, 48.85], [55.27, 25.2], [-17.44, 14.7], [-73.9, 40.7]];
    const routes = origins.slice(1).map(([lon, lat]) => {
      const start = vector(-8.5, 33.25), end = vector(lon, lat);
      const angle = Math.acos(Math.max(-1, Math.min(1, start.reduce((sum, value, i) => sum + value * end[i], 0))));
      return Array.from({length: 61}, (_, i): Vector => {
        const t = i / 60, lift = 1 + Math.sin(t * Math.PI) * .21;
        return start.map((value, j) => (value * Math.sin((1-t)*angle) + end[j] * Math.sin(t*angle)) / Math.sin(angle) * lift) as Vector;
      });
    });
    const resize = () => {
      const rect = canvas.getBoundingClientRect(); width = rect.width; height = rect.height;
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      canvas.width = width * dpr; canvas.height = height * dpr; context.setTransform(dpr, 0, 0, dpr, 0, 0);
      draw();
    };
    function draw() {
      if (!context || !width || !height) return;
      context.clearRect(0, 0, width, height);
      const cx = width * .5, cy = height * .49, radius = Math.min(width, height) * .34;
      const yaw = -.12 + Math.sin(elapsed * .000022) * .58, tilt = 22 * radians;
      const project = ([x, y, z]: Vector) => {
        const x1 = x * Math.cos(yaw) + z * Math.sin(yaw), z1 = z * Math.cos(yaw) - x * Math.sin(yaw);
        return [cx + x1 * radius, cy + (y * Math.cos(tilt) + z1 * Math.sin(tilt)) * radius, z1 * Math.cos(tilt) - y * Math.sin(tilt)];
      };
      for (const star of stars) { context.fillStyle = `rgba(149,208,193,${star.a})`; context.fillRect(star.x*width, star.y*height, 1, 1); }
      const glow = context.createRadialGradient(cx,cy,radius*.8,cx,cy,radius*1.5); glow.addColorStop(0,"rgba(39,165,139,.11)");glow.addColorStop(1,"rgba(7,31,39,0)");context.fillStyle=glow;context.fillRect(0,0,width,height);
      context.save(); context.translate(cx,cy); context.rotate(-.36);context.strokeStyle="rgba(92,172,151,.19)";context.lineWidth=.7;context.beginPath();context.ellipse(0,0,radius*1.29,radius*.39,0,0,Math.PI*2);context.stroke();context.restore();
      const sphere = context.createRadialGradient(cx-radius*.4,cy-radius*.45,0,cx,cy,radius);sphere.addColorStop(0,"#144a4f");sphere.addColorStop(.6,"#0b3038");sphere.addColorStop(1,"#061d25");context.fillStyle=sphere;context.beginPath();context.arc(cx,cy,radius,0,Math.PI*2);context.fill();context.strokeStyle="rgba(72,185,150,.32)";context.stroke();
      function line(points: Vector[], color: string, thickness: number) {
        if (!context) return;
        context.strokeStyle=color;context.lineWidth=thickness;context.beginPath();let previous=false;
        for (const point of points) { const [x,y,z]=project(point);if(z>0){ if(previous)context.lineTo(x,y);else context.moveTo(x,y);previous=true;}else previous=false; }context.stroke();
      }
      graticule.forEach(points=>line(points,"rgba(104,168,157,.13)",.65));
      coastlines.forEach(points=>line(points,"rgba(113,217,175,.75)",1));
      context.fillStyle="rgba(100,226,176,.55)";
      for (const points of coastlines) for (let i=0;i<points.length;i+=3){const [x,y,z]=project(points[i]);if(z>.08){context.beginPath();context.arc(x,y,.75+z*.4,0,Math.PI*2);context.fill();}}
      routes.forEach((points,index)=>{
        line(points,"rgba(69,206,155,.32)",.8);
        const current = points[Math.floor((elapsed*.008+index*13)%60)];const [x,y,z]=project(current);if(z>0){context.fillStyle="#bcffe0";context.shadowColor="#3edca7";context.shadowBlur=10;context.beginPath();context.arc(x,y,2.1,0,Math.PI*2);context.fill();context.shadowBlur=0;}
      });
      origins.forEach(([lon,lat],i)=>{const [x,y,z]=project(vector(lon,lat));if(z>0){context.fillStyle=i===0?"#d4fbe6":"#6bd7b1";context.beginPath();context.arc(x,y,i===0?4:2,0,Math.PI*2);context.fill();if(i===0){context.strokeStyle="rgba(101,225,173,.35)";context.beginPath();context.arc(x,y,9+(Math.sin(elapsed*.002)+1)*3,0,Math.PI*2);context.stroke();context.font="11px Arial";context.fillStyle="#bde8d8";context.fillText("El Jadida, Maroc",x+17,y-9);}}});
      context.strokeStyle="rgba(146,204,183,.2)";context.lineWidth=1;context.setLineDash([2,9]);context.beginPath();context.arc(cx,cy,radius*1.12,0,Math.PI*2);context.stroke();context.setLineDash([]);
    }
    function animate(time: number) {
      const delta = Math.min(time-lastTime, 40);lastTime=time;
      if(active && !document.hidden && !pauseRef.current && !media.matches && time-lastDraw>30){elapsed+=delta;draw();lastDraw=time;}
      frame=requestAnimationFrame(animate);
    }
    const resizeObserver = new ResizeObserver(resize);resizeObserver.observe(canvas);
    const observer = new IntersectionObserver(entries=>{active=entries[0].isIntersecting;});observer.observe(canvas);
    fetch("/data/coastlines.json",{signal:controller.signal}).then(r=>{if(!r.ok)throw Error("Map unavailable");return r.json();}).then((lines:number[][][])=>{coastlines=lines.map(line=>line.map(([lon,lat])=>vector(lon,lat)));draw();}).catch(()=>draw());
    media.addEventListener("change",draw);resize();frame=requestAnimationFrame(animate);
    return ()=>{controller.abort();cancelAnimationFrame(frame);resizeObserver.disconnect();observer.disconnect();media.removeEventListener("change",draw);};
  }, []);
  return <canvas ref={canvasRef} className="carbon-globe" aria-hidden="true" />;
}
